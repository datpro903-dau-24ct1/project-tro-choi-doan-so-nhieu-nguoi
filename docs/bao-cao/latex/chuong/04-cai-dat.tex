\begin{muctieu}
Mô tả môi trường triển khai thực tế, trình bày các đoạn mã nguồn cốt lõi giải
quyết bài toán đồng bộ hóa và phòng chống Deadlock, giao diện người dùng và
toàn bộ kết quả kiểm thử TDD.

\section{Các thành phần chính và mã nguồn cốt lõi}

\begin{lstlisting}[language=Java, caption={Thuật toán Lock Ordering chống Deadlock khi chuyển phòng (QuanLyPhongChoi.java)}, label={lst:lockorder}]
public KetQua<Void> chuyenPhong(String maCu, String maMoi, String ten) {
    var pCu = dsPhong.get(maCu.toUpperCase());
    var pMoi = dsPhong.get(maMoi.toUpperCase());
    if (pCu == null || pMoi == null) return new KetQua.ThatBai<>("PHONG_KHONG_TON_TAI", "...");

    // Acquire room locks in deterministic lexicographical order (Lock Ordering Rule - Ch4)
    PhongChoi khoa1 = maCu.compareTo(maMoi) < 0 ? pCu : pMoi;
    PhongChoi khoa2 = maCu.compareTo(maMoi) < 0 ? pMoi : pCu;

    synchronized (khoa1) {
        synchronized (khoa2) {
            pCu.xoaNguoiChoi(ten);
            var kq = pMoi.themNguoiChoi(ten);
            if (kq instanceof KetQua.ThanhCong) {
                nguoiChoiThuocPhong.put(ten, maMoi);
            }
            return kq;
        }
    }
}
\end{lstlisting}

\begin{lstlisting}[language=Java, caption={Xử lý đồng bộ ván chơi và chốt người thắng đầu tiên (Game.java)}, label={lst:game}]
public synchronized KetQua<LuotDoan> doanSo(String ten, int soDoan) {
    if (trangThai != TrangThaiVan.PLAYING) {
        return new KetQua.ThatBai<>("VAN_DA_KET_THUC", "Game is not in PLAYING state");
    }
    GoiY goiY = GoiY.soSanh(soDoan, soBiMat);
    LuotDoan luot = new LuotDoan(ten, soDoan, LocalDateTime.now(), goiY);
    dsLuotDoan.add(luot);
    if (goiY == GoiY.CORRECT) {
        this.nguoiThang = ten;
        this.trangThai = TrangThaiVan.FINISHED; // Lock winner and close round immediately
    }
    return new KetQua.ThanhCong<>(luot, "Valid guess");
}

\section{Giao diện máy khách ClientUI}

Giao diện người dùng \code{ClientUI} được lập trình bằng thư viện Swing tiêu
chuẩn. Cửa sổ bao gồm các trường nhập địa chỉ máy chủ, cổng, tên người chơi,
khu vực nút điều khiển (SCORE, HISTORY, STATS, QUIT), ô nhập số đoán và vùng
nhật ký (\code{JTextArea} bọc trong \code{JScrollPane}). Đặc biệt, luồng đọc
mạng từ máy chủ được tách biệt trên một \code{Thread} riêng biệt, giúp toàn bộ
tương tác người dùng luôn mượt mà và không bao giờ bị đơ giao diện.
