\section{Các thành phần chính}

\subsection{Bản ghi lượt đoán bất biến và đồng bộ hóa ván chơi (\code{F03}, \code{F04}, \code{F19})}

Mã nguồn~\ref{lst:luotdoan} minh họa bản ghi \code{LuotDoan} sử dụng
\code{record} và hàm dựng gọn \en{(compact constructor)} của Chương 6 để kiểm
tra tính hợp lệ và chuẩn hóa tên người chơi ngay khi khởi tạo.

\begin{lstlisting}[language=Java, caption={Cài đặt record bất biến LuotDoan kèm compact constructor.}, label={lst:luotdoan}]
package doanso.domain;

import java.time.LocalDateTime;

public record LuotDoan(String nguoiChoi, int soDoan, LocalDateTime thoiDiem, GoiY ketQuaGoiY) {

    // Validate fields and normalize player name inside compact constructor
    public LuotDoan {
        if (nguoiChoi == null || nguoiChoi.isBlank()) {
            throw new IllegalArgumentException("Player name must not be blank");
        }
        if (thoiDiem == null) {
            throw new IllegalArgumentException("Timestamp must not be null");
        }
        if (ketQuaGoiY == null) {
            throw new IllegalArgumentException("Hint result must not be null");
        }
        nguoiChoi = nguoiChoi.strip();
    }
}
\end{lstlisting}

\subsection{Thống kê dữ liệu bằng đường ống Streams API (\code{F16})}

Mã nguồn~\ref{lst:thongke} minh họa lớp \code{ThongKeService} sử dụng
\code{Stream}, \code{filter}, \code{mapToInt}, \code{average} và
\code{Collectors.groupingBy} để tính số lần đoán trung bình mỗi ván đã hoàn
tất và thống kê số lượt đoán của từng người chơi mà không dùng vòng lặp thủ
công.

\begin{lstlisting}[language=Java, caption={Thống kê số lượt đoán trung bình bằng Java Streams API trong ThongKeService.java.}, label={lst:thongke}]
// Compute average guesses across finished games that have a winner
public double tinhTrungBinhSoLuotMoiVan(Collection<Game> danhSachVan) {
    if (danhSachVan == null || danhSachVan.isEmpty()) {
        return 0.0;
    }
    return danhSachVan.stream()
        .filter(Objects::nonNull)
        .filter(Game::daCoNguoiThang)
        .mapToInt(Game::getSoLuotDoan)
        .average()
        .orElse(0.0);
}

// Group and count total guesses per player using Collectors.groupingBy
public Map<String, Long> thongKeSoLuotTheoTungNguoi(KhoLichSuDoan khoLichSu) {
    if (khoLichSu == null) {
        return Map.of();
    }
    return khoLichSu.layTatCaLuotDoan().stream()
        .filter(Objects::nonNull)
        .collect(Collectors.groupingBy(LuotDoan::nguoiChoi, Collectors.counting()));
}
\end{lstlisting}

\subsection{Sắp xếp khóa chống bế tắc khi chuyển phòng chơi (\code{F21})}

Mã nguồn~\ref{lst:lockorder} minh họa phương thức \code{vaoPhong} trong
\code{QuanLyPhongChoi.java}, áp dụng quy tắc giành khóa theo thứ tự từ điển
cố định của công thức~\eqref{eq:lockorder} để triệt tiêu hoàn toàn nguy cơ
\code{deadlock} khi nhiều luồng chuyển phòng chéo nhau cùng lúc.

\begin{lstlisting}[language=Java, caption={Giành khóa theo thứ tự từ điển cố định chống Deadlock trong QuanLyPhongChoi.java.}, label={lst:lockorder}]
// Acquire room locks in deterministic lexicographical order to prevent deadlock
PhongChoi khoa1 = maCu.compareTo(maMoi) < 0 ? pCu : pMoi;
PhongChoi khoa2 = maCu.compareTo(maMoi) < 0 ? pMoi : pCu;

synchronized (khoa1) {
    synchronized (khoa2) {
        pCu.xoaNguoiChoi(tenChuan);
        if (pCu.getTrangThaiVanHienTai() == TrangThaiVan.PLAYING) {
            soOnline.broadcastTrongTap(
                pCu.getDanhSachNguoiChoi(), "INFO NGUOI_ROI_PHONG " + tenChuan
            );
        }
        KetQua<Void> kqVao = pMoi.themNguoiChoi(tenChuan);
        if (kqVao instanceof KetQua.ThanhCong) {
            nguoiChoiThuocPhong.put(tenChuan, maMoi);
        }
        return kqVao;
    }
}
\end{lstlisting}

\section{Giao diện và vận hành thực tế}

Giao diện máy khách \code{ClientUI} được xây dựng bằng Java Swing
(\code{JFrame}), bố trí thành 3 vùng rõ ràng: thanh cấu hình kết nối ở phía
trên (\code{Host}, \code{Port}, \code{Tên}, nút \code{Kết nối}), khung nhật
ký ván chơi có thanh cuộn ở chính giữa (\code{JScrollPane} chứa
\code{JTextArea}), và thanh điều khiển ở phía dưới (\code{Số đoán}, nút
\code{Đoán}, \code{Mở ván}, \code{Bảng điểm}, \code{Lịch sử}, \code{Thống kê},
\code{Thoát}). Ngay khi kết nối thành công, \code{ClientUI} khởi chạy một
luồng nền riêng (\code{LuongDocClient}) để lắng nghe thông điệp từ máy chủ và
đẩy cập nhật giao diện về luồng sự kiện Swing qua
\code{SwingUtilities.invokeLater}, giữ cho cửa sổ luôn mượt mà.
